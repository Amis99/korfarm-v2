\section*{여기서 할 수 있는 일}

\begin{itemize}[leftmargin=1.2em]
  \item 본인 기관의 오늘 가입자·활성 학생·누적 학생·활성 학부모를 한 눈에 파악
  \item ``승인 대기'' 카드 클릭 → 가입 승인 페이지로 이동
  \item ``학부모 연결 대기'' 카드 클릭 → 학부모 관리로 이동
  \item ``최근 7일 시험 응시'' 카드 클릭 → 테스트 관리로 이동
  \item ``빠른 이동'' / ``관리 메뉴'' 버튼 → 자주 쓰는 화면으로 한 번에
\end{itemize}

\section*{자주 쓰는 흐름}

\begin{enumerate}
  \item 출근 → 로그인 → 대시보드에서 \textbf{오늘 가입자}와 \textbf{승인 대기} 확인
  \item 승인 대기가 0 이 아니면 카드 클릭 → 가입 승인 페이지에서 처리 (Chapter 2)
  \item 학부모 연결 대기가 있으면 학부모 관리에서 매칭 (Chapter 5)
  \item 학습 진척이 궁금하면 ``학습 계획표 / 통합 성적표'' 로 이동
\end{enumerate}

\begin{tipbox}{알아두면 좋은 점}
\begin{itemize}[leftmargin=1.2em]
  \item 모든 숫자는 \textbf{본인 기관 소속 사용자만} 집계됩니다 (본사가 보는 전사 숫자와 다름)
  \item 월 사용료가 미결제로 정지된 경우, 화면 상단에 빨간 배너가 떠 결제 페이지로 안내합니다 — 이때만큼은 결제부터
  \item 결제 마감 7일 이내일 때 노란 배너 — 임박 안내
\end{itemize}
\end{tipbox}
