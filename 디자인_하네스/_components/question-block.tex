% 타입별 배지 매크로 (@ 없이 — 인용부가 \makeatletter 모드가 아니어도 안전)
\newcommand{\kfQnumDraw}[2]{%
  \tikz[baseline=(qn.base)]{%
    \node[draw=#1, fill=#1!10, text=#1,
          rounded corners=3pt, minimum width=22pt,
          inner xsep=4pt, inner ysep=2pt,
          font=\small\bfseries]
          (qn) {#2};%
  }%
}

\newcommand{\kfQNumObj}[1]{\kfQnumDraw{kfPrimary}{#1}}
\newcommand{\kfQNumShort}[1]{\kfQnumDraw{kfAreaGrammar}{#1}}
\newcommand{\kfQNumDesc}[1]{\kfQnumDraw{kfAccent}{#1}}
\renewcommand{\kfQNum}[1]{\kfQNumObj{#1}}

% 유형 문자열로 배지 색 선택
\def\kfTypeShort{단답형}
\def\kfTypeDesc{서술형}

\newcommand{\kfQNumByType}[2]{%
  \def\kfCurType{#1}%
  \ifx\kfCurType\kfTypeShort
    \kfQNumShort{#2}%
  \else
    \ifx\kfCurType\kfTypeDesc
      \kfQNumDesc{#2}%
    \else
      \kfQNumObj{#2}%
    \fi
  \fi
}

% 무단절 + 타입별 번호 배지
% v14: tcolorbox(blanker, breakable=false)로 회귀 — 자동 fits 검사
%      multicols* 시작 직전 \needspace{0.45\textheight}로 컬럼 측정 정확성 보장
%      lrbox 측정 + \columnbreak 로직은 너무 보수적이라 단 바꿈 과다 발생 (#34) → 폐기
% PAGE_BREAK_POLICY.md §2.6, §5
\newenvironment{kfQBlock}[2]%
  {\par\smallskip%
   \begin{tcolorbox}[
     blanker, breakable=false, parbox=false,
     left=0pt, right=0pt, top=0pt, bottom=0pt,
     before skip=2pt, after skip=4pt
   ]%
   \noindent\kfQNumByType{#2}{#1}\hspace{4pt}%
   \begin{minipage}[t]{\dimexpr\linewidth-34pt\relax}%
       \ignorespaces%
  }%
  {\end{minipage}\end{tcolorbox}}

% 문제 발문
\newcommand{\kfQStem}[1]{%
  \par\noindent#1\par\vspace{2pt}%
}
