%   \kfAreaStart{어휘}{kfAreaVocab}{이번 영역에서는 ... 학습합니다.}
% 또는 시리즈 단축 매크로:
%   \kfStartVocab{이번 영역에서는 핵심 어휘를 익힙니다.}
% ============================================================

\newcommand{\kfAreaStart}[3]{%
  \clearpage%
  \thispagestyle{fancy}%
  \vspace*{2mm}%
  \noindent
  \begin{tikzpicture}
    \node[anchor=west, inner sep=0pt] (bar) at (0,0) {\color{#2}\rule{5pt}{26pt}};
    \node[anchor=west, inner sep=0pt, xshift=8pt] at (bar.east) {%
      {\LARGE\bfseries\color{#2} #1}%
    };
  \end{tikzpicture}%
  \par\vspace{4pt}
  \noindent\hspace{14pt}%
  \begin{minipage}{\dimexpr\linewidth-14pt\relax}%
    \footnotesize\color{kfMute} #3
  \end{minipage}%
  \par\vspace{4pt}%
  {\color{#2!60}\hrule height 1.0pt}%
  \par\vspace{6pt}\nopagebreak%
}

% 시리즈/영역 단축 매크로
